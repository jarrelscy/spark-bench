import json
from fractions import Fraction

STATES = ("NEW", "AWAITING_PAYMENT", "PARTIALLY_PAID", "PAID",
          "SHIPPED", "DELIVERED", "CANCELLED", "CLOSED")


def _to_cents(dec_str):
    """Convert a decimal string like '19.99' exactly to integer cents."""
    s = str(dec_str).strip()
    neg = False
    if s.startswith("-"):
        neg = True
        s = s[1:]
    if "." in s:
        whole, frac = s.split(".", 1)
    else:
        whole, frac = s, ""
    if whole == "":
        whole = "0"
    # exactly two decimals per spec; tolerate missing/extra digits safely
    frac = (frac + "00")[:2]
    cents = int(whole) * 100 + int(frac)
    return -cents if neg else cents


def _round_half_up(frac):
    """Round a Fraction half up (towards +infinity for .5) to an integer."""
    if isinstance(frac, int):
        return frac
    num, den = frac.numerator, frac.denominator
    if den == 1:
        return num
    # floor division then compare remainder to half the denominator
    q, r = divmod(num, den)
    if 2 * r >= den:
        q += 1
    return q


class _OrderState:
    """Mutable internal state used for both the main run and the replay."""

    def __init__(self, order):
        self.order_id = order.get("id", "")
        self.lines = []
        for ln in order.get("lines", []) or []:
            qty = int(ln.get("qty", 0))
            price_cents = _to_cents(ln.get("price", "0.00"))
            self.lines.append({
                "sku": ln.get("sku", ""),
                "qty": qty,
                "price": price_cents,
                "amount": qty * price_cents,
            })
        coupon = order.get("coupon")
        self.coupon = coupon if isinstance(coupon, dict) else None

        # money
        self.subtotal = sum(l["amount"] for l in self.lines)
        self.discount = self._compute_discount()
        self.total = self.subtotal - self.discount

        # discount shares per line (exact, sum == discount)
        self.line_discounts = self._compute_shares()

        # state machine
        self.state = "NEW"
        self.paid = 0
        self.refunded = 0
        self.delivered_ts = None
        self.returned = {}          # sku -> returned qty
        self.line_refunded = {}     # line index -> refunded cents so far

    # ---- money helpers -------------------------------------------------

    def _compute_discount(self):
        if self.coupon is None or self.subtotal == 0:
            return 0
        percent = int(self.coupon.get("percent", 0))
        cap = _to_cents(self.coupon.get("cap", "0.00"))
        raw = Fraction(self.subtotal * percent, 100)
        disc = _round_half_up(raw)
        if disc > cap:
            disc = cap
        if disc < 0:
            disc = 0
        if disc > self.subtotal:
            disc = self.subtotal
        return disc

    def _compute_shares(self):
        n = len(self.lines)
        shares = [0] * n
        if self.discount == 0 or self.subtotal == 0:
            return shares
        acc = 0
        for i, ln in enumerate(self.lines):
            if i == n - 1:
                shares[i] = self.discount - acc
            else:
                shares[i] = _round_half_up(
                    Fraction(self.discount * ln["amount"], self.subtotal))
                acc += shares[i]
        return shares

    def line_net_amount(self, idx):
        return self.lines[idx]["amount"] - self.line_discounts[idx]

    def unit_net(self, idx):
        """Exact fractional unit net price of a line."""
        net = self.line_net_amount(idx)
        return Fraction(net, self.lines[idx]["qty"])

    # ---- event application ----------------------------------------------

    def _find_line(self, sku):
        for i, ln in enumerate(self.lines):
            if ln["sku"] == sku:
                return i
        return None

    def apply(self, ev):
        """Apply one event dict. Returns True if applied, False if rejected."""
        etype = ev.get("type")
        ts = ev.get("ts", 0)

        if etype == "place":
            if self.state != "NEW":
                return False
            if not self.lines:
                return False
            if self.total == 0:
                self.state = "PAID"
            else:
                self.state = "AWAITING_PAYMENT"
            return True

        if etype == "pay":
            if self.state not in ("AWAITING_PAYMENT", "PARTIALLY_PAID"):
                return False
            if self.total == 0:
                return False
            try:
                amt = _to_cents(ev.get("amount", "0.00"))
            except Exception:
                return False
            if amt <= 0:
                return False
            if self.paid + amt > self.total:
                return False
            self.paid += amt
            if self.paid == self.total:
                self.state = "PAID"
            else:
                self.state = "PARTIALLY_PAID"
            return True

        if etype == "ship":
            if self.state != "PAID":
                return False
            self.state = "SHIPPED"
            return True

        if etype == "deliver":
            if self.state != "SHIPPED":
                return False
            self.state = "DELIVERED"
            self.delivered_ts = ts
            return True

        if etype == "cancel":
            if self.state in ("NEW", "AWAITING_PAYMENT"):
                self.state = "CANCELLED"
                return True
            if self.state in ("PARTIALLY_PAID", "PAID"):
                # refund everything paid so far
                self.refunded += self.paid - self.refunded
                self.state = "CANCELLED"
                return True
            return False

        if etype == "return":
            if self.state != "DELIVERED":
                return False
            if self.delivered_ts is None or (ts - self.delivered_ts) > 30:
                return False
            sku = ev.get("sku")
            try:
                qty = int(ev.get("qty", 0))
            except Exception:
                return False
            idx = self._find_line(sku)
            if idx is None:
                return False
            if qty < 1:
                return False
            ln = self.lines[idx]
            already = self.returned.get(sku, 0)
            if already + qty > ln["qty"]:
                return False

            # refund for this return, capped at remaining line net amount
            refund = _round_half_up(self.unit_net(idx) * qty)
            net = self.line_net_amount(idx)
            already_refunded = self.line_refunded.get(idx, 0)
            remaining = net - already_refunded
            if remaining < 0:
                remaining = 0
            if refund > remaining:
                refund = remaining
            # final return of the line: refund exactly the remainder
            if already + qty == ln["qty"]:
                refund = remaining
            if refund > 0:
                self.refunded += refund
                self.line_refunded[idx] = already_refunded + refund

            self.returned[sku] = already + qty

            # if every unit of every line is returned -> CLOSED
            all_returned = True
            for l2 in self.lines:
                if self.returned.get(l2["sku"], 0) < l2["qty"]:
                    all_returned = False
                    break
            if all_returned:
                self.state = "CLOSED"
            return True

        if etype == "close":
            if self.state != "DELIVERED":
                return False
            self.state = "CLOSED"
            return True

        return False


def run_trace(order, events):
    # ---- first run: sort events by ts (stable), detect duplicates --------
    ordered = sorted(list(events), key=lambda e: e.get("ts", 0))
    seen_ids = set()
    audit = []
    duplicates = 0
    applied_count = 0
    rejected_count = 0

    st = _OrderState(order)

    for ev in ordered:
        eid = ev.get("id")
        if eid in seen_ids:
            duplicates += 1
            continue
        seen_ids.add(eid)
        ok = st.apply(ev)
        if ok:
            applied_count += 1
            audit.append({
                "id": eid,
                "type": ev.get("type"),
                "ts": ev.get("ts", 0),
                "result": "applied",
                "state_after": st.state,
            })
        else:
            rejected_count += 1
            audit.append({
                "id": eid,
                "type": ev.get("type"),
                "ts": ev.get("ts", 0),
                "result": "rejected",
                "state_after": st.state,
            })

    # ---- replay: fresh order, re-apply only applied audit entries --------
    st2 = _OrderState(order)
    replay_ok = True
    for entry in audit:
        if entry["result"] != "applied":
            continue
        # find the original event dict by id (ids are unique among applied)
        orig = None
        for ev in ordered:
            if ev.get("id") == entry["id"]:
                orig = ev
                break
        if orig is None:
            replay_ok = False
            break
        ok = st2.apply(orig)
        if not ok:
            replay_ok = False
            break
    if replay_ok:
        if st2.state != st.state:
            replay_ok = False
        elif st2.paid != st.paid:
            replay_ok = False
        elif st2.refunded != st.refunded:
            replay_ok = False

    # ---- returned map: only skus with qty > 0 -----------------------------
    returned_out = {sku: q for sku, q in st.returned.items() if q > 0}

    return {
        "state": st.state,
        "subtotal": st.subtotal,
        "discount": st.discount,
        "total": st.total,
        "paid": st.paid,
        "refunded": st.refunded,
        "applied": applied_count,
        "rejected": rejected_count,
        "duplicates": duplicates,
        "returned": returned_out,
        "audit_len": len(audit),
        "replay_ok": replay_ok,
    }


if __name__ == "__main__":
    import sys
    data = json.load(sys.stdin)
    result = run_trace(data["order"], data["events"])
    print(json.dumps(result))
