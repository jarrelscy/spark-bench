"""sheet.py — spreadsheet formula engine (standard library only, Python 3.11)."""

import math
import re
from decimal import Decimal, ROUND_HALF_UP, localcontext


# ---------------------------------------------------------------------------
# Reference parsing
# ---------------------------------------------------------------------------

_REF_RE = re.compile(r'^([A-Za-z]{1,2})([0-9]{1,4})$')


def _col_to_num(letters):
    """Excel-style column letters -> 1-based index (A=1..Z=26, AA=27..ZZ=702)."""
    n = 0
    for ch in letters:
        n = n * 26 + (ord(ch) - ord('A') + 1)
    return n


def _num_to_col(n):
    """1-based column index -> Excel-style letters."""
    s = ''
    while n > 0:
        n, r = divmod(n - 1, 26)
        s = chr(ord('A') + r) + s
    return s


def parse_ref(ref):
    """Parse a cell reference string -> (row, col) 1-based, or None if invalid."""
    if not isinstance(ref, str):
        return None
    m = _REF_RE.match(ref)
    if not m:
        return None
    letters = m.group(1).upper()
    row = int(m.group(2))
    if row < 1 or row > 9999:
        return None
    col = _col_to_num(letters)
    if col > 702:  # ZZ
        return None
    return (row, col)


def canonical_ref(ref):
    """Canonical (upper-case) form of a valid ref, or None if invalid."""
    rc = parse_ref(ref)
    if rc is None:
        return None
    row, col = rc
    return _num_to_col(col) + str(row)


# ---------------------------------------------------------------------------
# Tokenizer
# ---------------------------------------------------------------------------

_TOKEN_RE = re.compile(r"""
    (?P<ws>\s+)
  | (?P<number>(?:[0-9]+(?:\.[0-9]*)?|\.[0-9]+)(?:[eE][+-]?[0-9]+)?)
  | (?P<string>"(?:[^"]|"")*")
  | (?P<ident>[A-Za-z_][A-Za-z0-9_.]*)
  | (?P<op><>|<=|>=|[=<>&+\-*/^(),:!])
""", re.VERBOSE)


class Token:
    __slots__ = ('kind', 'value')

    def __init__(self, kind, value):
        self.kind = kind
        self.value = value

    def __repr__(self):
        return 'Token(%r, %r)' % (self.kind, self.value)


def tokenize(text):
    """Tokenize expression text -> list of Tokens, or None on bad char."""
    toks = []
    pos = 0
    n = len(text)
    while pos < n:
        m = _TOKEN_RE.match(text, pos)
        if m is None:
            return None
        pos = m.end()
        kind = m.lastgroup
        val = m.group()
        if kind == 'ws':
            continue
        if kind == 'number':
            toks.append(Token('num', float(val)))
        elif kind == 'string':
            toks.append(Token('str', val[1:-1].replace('""', '"')))
        elif kind == 'ident':
            toks.append(Token('ident', val))
        else:  # op
            toks.append(Token('op', val))
    return toks


# ---------------------------------------------------------------------------
# AST nodes (tuples: (tag, ...))
#   ('num', float)
#   ('str', str)
#   ('bool', bool)
#   ('ref', row, col, canonical)
#   ('range', r1, c1, r2, c2)
#   ('name', text)                bare identifier -> #NAME?
#   ('call', NAME_upper, [args])
#   ('bin', op, left, right)
#   ('un', op, operand)
# ---------------------------------------------------------------------------

_CMP_OPS = ('=', '<>', '<', '>', '<=', '>=')


class Parser:
    def __init__(self, toks):
        self.toks = toks
        self.i = 0
        self.n = len(toks)

    def peek(self):
        return self.toks[self.i] if self.i < self.n else None

    def advance(self):
        t = self.toks[self.i]
        self.i += 1
        return t

    def expect_op(self, op):
        t = self.peek()
        if t is not None and t.kind == 'op' and t.value == op:
            return self.advance()
        return None

    def parse(self):
        node = self.parse_cmp()
        if self.peek() is not None:
            return None  # trailing junk
        return node

    def parse_cmp(self):
        node = self.parse_concat()
        while True:
            t = self.peek()
            if t is not None and t.kind == 'op' and t.value in _CMP_OPS:
                self.advance()
                rhs = self.parse_concat()
                if rhs is None:
                    return None
                node = ('bin', t.value, node, rhs)
            else:
                return node

    def parse_concat(self):
        node = self.parse_add()
        while True:
            t = self.peek()
            if t is not None and t.kind == 'op' and t.value == '&':
                self.advance()
                rhs = self.parse_add()
                if rhs is None:
                    return None
                node = ('bin', '&', node, rhs)
            else:
                return node

    def parse_add(self):
        node = self.parse_mul()
        while True:
            t = self.peek()
            if t is not None and t.kind == 'op' and t.value in ('+', '-'):
                self.advance()
                rhs = self.parse_mul()
                if rhs is None:
                    return None
                node = ('bin', t.value, node, rhs)
            else:
                return node

    def parse_mul(self):
        node = self.parse_unary()
        while True:
            t = self.peek()
            if t is not None and t.kind == 'op' and t.value in ('*', '/'):
                self.advance()
                rhs = self.parse_unary()
                if rhs is None:
                    return None
                node = ('bin', t.value, node, rhs)
            else:
                return node

    def parse_unary(self):
        t = self.peek()
        if t is not None and t.kind == 'op' and t.value in ('-', '+'):
            self.advance()
            operand = self.parse_unary()
            if operand is None:
                return None
            return ('un', t.value, operand)
        return self.parse_power()

    def parse_power(self):
        base = self.parse_primary()
        if base is None:
            return None
        t = self.peek()
        if t is not None and t.kind == 'op' and t.value == '^':
            self.advance()
            exp = self.parse_unary()  # right-associative, allows 2^-1
            if exp is None:
                return None
            return ('bin', '^', base, exp)
        return base

    def parse_primary(self):
        t = self.peek()
        if t is None:
            return None
        if t.kind == 'num':
            self.advance()
            return ('num', t.value)
        if t.kind == 'str':
            self.advance()
            return ('str', t.value)
        if t.kind == 'op' and t.value == '(':
            self.advance()
            node = self.parse_cmp()
            if node is None:
                return None
            if self.expect_op(')') is None:
                return None
            return node
        if t.kind == 'ident':
            return self.parse_ident()
        return None

    def parse_ident(self):
        t = self.advance()  # an 'ident' token
        text = t.value
        up = text.upper()
        # Function call?
        nxt = self.peek()
        if nxt is not None and nxt.kind == 'op' and nxt.value == '(':
            self.advance()
            args = []
            first = self.peek()
            if first is not None and first.kind == 'op' and first.value == ')':
                self.advance()
                return ('call', up, args)
            while True:
                arg = self.parse_arg()
                if arg is None:
                    return None
                args.append(arg)
                nxt = self.peek()
                if nxt is not None and nxt.kind == 'op' and nxt.value == ',':
                    self.advance()
                    continue
                if nxt is not None and nxt.kind == 'op' and nxt.value == ')':
                    self.advance()
                    return ('call', up, args)
                return None
        if up == 'TRUE':
            return ('bool', True)
        if up == 'FALSE':
            return ('bool', False)
        rc = parse_ref(text)
        if rc is not None:
            row, col = rc
            return ('ref', row, col, canonical_ref(text))
        # Range A1:B3?
        nxt = self.peek()
        if nxt is not None and nxt.kind == 'op' and nxt.value == ':':
            # lookahead: after ':' must be an ident shaped like a ref
            save = self.i
            self.advance()
            t2 = self.peek()
            if t2 is not None and t2.kind == 'ident':
                rc2 = parse_ref(t2.value)
                if rc2 is not None:
                    self.advance()
                    rc1 = parse_ref(text)
                    r1, c1 = rc1
                    r2, c2 = rc2
                    return ('range', min(r1, r2), min(c1, c2),
                            max(r1, r2), max(c1, c2))
            self.i = save
        return ('name', text)

    def parse_arg(self):
        """An argument: expression, or a range."""
        # Range as argument: ident ':' ident where both are refs.
        t = self.peek()
        if t is not None and t.kind == 'ident':
            rc1 = parse_ref(t.value)
            if rc1 is not None:
                nxt = self.toks[self.i + 1] if self.i + 1 < self.n else None
                if nxt is not None and nxt.kind == 'op' and nxt.value == ':':
                    t2 = self.toks[self.i + 2] if self.i + 2 < self.n else None
                    if t2 is not None and t2.kind == 'ident':
                        rc2 = parse_ref(t2.value)
                        if rc2 is not None:
                            self.advance()  # ident
                            self.advance()  # ':'
                            self.advance()  # ident2
                            r1, c1 = rc1
                            r2, c2 = rc2
                            return ('range', min(r1, r2), min(c1, c2),
                                    max(r1, r2), max(c1, c2))
        return self.parse_cmp()


def parse_formula(expr):
    """Parse the text after '=' -> AST or None on syntax error."""
    toks = tokenize(expr)
    if toks is None or not toks:
        return None
    return Parser(toks).parse()


# ---------------------------------------------------------------------------
# Evaluation
# ---------------------------------------------------------------------------

DIV0 = '#DIV/0!'
VALUE = '#VALUE!'
REF = '#REF!'
NAME = '#NAME?'
PARSE = '#PARSE!'
CYCLE = '#CYCLE!'

_ERROR_SET = frozenset((DIV0, VALUE, REF, NAME, PARSE, CYCLE))


def is_error(v):
    return isinstance(v, str) and v in _ERROR_SET


def to_number(v):
    """Arithmetic coercion -> float, or an error string."""
    if v is None:
        return 0.0
    if isinstance(v, bool):
        return 1.0 if v else 0.0
    if isinstance(v, float):
        return v
    if isinstance(v, str):
        return VALUE
    return VALUE


def to_text(v):
    """Text coercion -> str."""
    if v is None:
        return ''
    if isinstance(v, bool):
        return 'TRUE' if v else 'FALSE'
    if isinstance(v, float):
        return format_number(v)
    return v


def format_number(f):
    """Number -> text per spec: integral and |x| < 1e15 -> no decimal point."""
    if isinstance(f, bool):
        return 'TRUE' if f else 'FALSE'
    if math.isfinite(f) and f == int(f) and abs(f) < 1e15:
        return str(int(f))
    return repr(f)


def compare(op, a, b):
    """Comparison -> bool or error string."""
    # Empty participates as number 0.0 or text "" depending on the other side.
    if a is None:
        if isinstance(b, str) and not isinstance(b, bool) and not is_error(b):
            a = ''
        elif b is None:
            a = 0.0
        else:
            a = 0.0
    if b is None:
        if isinstance(a, str) and not is_error(a):
            b = ''
        else:
            b = 0.0
    a_is_text = isinstance(a, str) and not is_error(a)
    b_is_text = isinstance(b, str) and not is_error(b)
    if a_is_text and b_is_text:
        x, y = a.upper(), b.upper()
    elif a_is_text or b_is_text:
        return VALUE
    else:
        # bools coerce to numbers
        x = 1.0 if a is True else (0.0 if a is False else a)
        y = 1.0 if b is True else (0.0 if b is False else b)
        if not (isinstance(x, float) and isinstance(y, float)):
            return VALUE
    if op == '=':
        return x == y
    if op == '<>':
        return x != y
    if op == '<':
        return x < y
    if op == '>':
        return x > y
    if op == '<=':
        return x <= y
    if op == '>=':
        return x >= y
    return VALUE


def _round_half_away(x, digits):
    """Round half away from zero to `digits` decimals, exact on decimal ties."""
    d = Decimal(repr(x))
    q = Decimal(1).scaleb(-digits)  # 10^-digits
    with localcontext() as ctx:
        ctx.prec = 60
        r = d.quantize(q, rounding=ROUND_HALF_UP)
    return float(r)


class Evaluator:
    """Evaluates an AST for one cell against the sheet.

    sheet must provide: get_value(row, col) -> evaluated value of a cell,
    and is_known_formula(row, col) -> bool (cell holds a formula).
    """

    def __init__(self, sheet, row, col):
        self.sheet = sheet
        self.row = row
        self.col = col

    def eval(self, node, allow_range=False):
        tag = node[0]
        if tag == 'num':
            return node[1]
        if tag == 'str':
            return node[1]
        if tag == 'bool':
            return node[1]
        if tag == 'ref':
            _, r, c, canon = node
            if r > 9999 or c > 702:
                return REF
            return self.sheet.get_value(r, c)
        if tag == 'range':
            if not allow_range:
                return VALUE
            return node  # caller handles ranges
        if tag == 'name':
            return NAME
        if tag == 'bin':
            return self.eval_bin(node)
        if tag == 'un':
            return self.eval_un(node)
        if tag == 'call':
            return self.eval_call(node)
        return VALUE

    def eval_bin(self, node):
        _, op, ln, rn = node
        a = self.eval(ln)
        if is_error(a):
            return a
        b = self.eval(rn)
        if is_error(b):
            return b
        if op in _CMP_OPS:
            return compare(op, a, b)
        if op == '&':
            return to_text(a) + to_text(b)
        x = to_number(a)
        if is_error(x):
            return x
        y = to_number(b)
        if is_error(y):
            return y
        if op == '+':
            r = x + y
        elif op == '-':
            r = x - y
        elif op == '*':
            r = x * y
        elif op == '/':
            if y == 0.0:
                return DIV0
            r = x / y
        elif op == '^':
            if x == 0.0 and y < 0:
                return DIV0
            try:
                r = x ** y
            except (OverflowError, ZeroDivisionError):
                return VALUE
            if isinstance(r, complex):
                return VALUE
        else:
            return VALUE
        if not math.isfinite(r):
            return VALUE
        return float(r)

    def eval_un(self, node):
        _, op, sub = node
        v = self.eval(sub)
        if is_error(v):
            return v
        x = to_number(v)
        if is_error(x):
            return x
        r = -x if op == '-' else x
        if not math.isfinite(r):
            return VALUE
        return float(r)

    # -- function calls ----------------------------------------------------

    def eval_call(self, node):
        _, name, args = node
        if name not in _FUNCS:
            return NAME
        return _FUNCS[name](self, args)

    def _iter_values(self, args, skip_errors=False, count_only_numbers=False):
        """Yield (value, came_from_range) for all cells/args in order.

        Returns an error string immediately if one is met (unless
        skip_errors / count_only_numbers).
        """
        for a in args:
            if a[0] == 'range':
                _, r1, c1, r2, c2 = a
                if r1 > 9999 or r2 > 9999 or c1 > 702 or c2 > 702:
                    if not skip_errors and not count_only_numbers:
                        return REF
                    continue
                for r in range(r1, r2 + 1):
                    for c in range(c1, c2 + 1):
                        v = self.sheet.get_value(r, c)
                        yield v, True
            else:
                v = self.eval(a, allow_range=False)
                yield v, False

    def _collect_numbers(self, args, func_name):
        """Shared logic for SUM/AVERAGE/MIN/MAX/COUNT."""
        nums = []
        for a in args:
            if a[0] == 'range':
                _, r1, c1, r2, c2 = a
                if r1 > 9999 or r2 > 9999 or c1 > 702 or c2 > 702:
                    return REF
                for r in range(r1, r2 + 1):
                    for c in range(c1, c2 + 1):
                        v = self.sheet.get_value(r, c)
                        if is_error(v):
                            if func_name == 'COUNT':
                                continue
                            return v
                        if isinstance(v, float) and not isinstance(v, bool):
                            nums.append(v)
                        # text, bool, empty in ranges: skipped
            else:
                v = self.eval(a, allow_range=False)
                if is_error(v):
                    if func_name == 'COUNT':
                        continue
                    return v
                if func_name == 'COUNT':
                    if isinstance(v, float) and not isinstance(v, bool):
                        nums.append(v)
                    elif v is None:
                        nums.append(0.0)  # empty expression counts as 0
                    elif isinstance(v, bool):
                        nums.append(1.0 if v else 0.0)
                    elif isinstance(v, str):
                        pass  # text direct arg: not counted
                    continue
                if isinstance(v, bool):
                    nums.append(1.0 if v else 0.0)
                elif isinstance(v, float):
                    nums.append(v)
                elif v is None:
                    nums.append(0.0)
                else:  # text direct argument
                    return VALUE
        return nums


def _fn_sum(ev, args):
    nums = ev._collect_numbers(args, 'SUM')
    if is_error(nums):
        return nums
    r = math.fsum(nums)
    if not math.isfinite(r):
        return VALUE
    return float(r)


def _fn_average(ev, args):
    nums = ev._collect_numbers(args, 'AVERAGE')
    if is_error(nums):
        return nums
    if not nums:
        return DIV0
    r = math.fsum(nums) / len(nums)
    if not math.isfinite(r):
        return VALUE
    return float(r)


def _fn_min(ev, args):
    nums = ev._collect_numbers(args, 'MIN')
    if is_error(nums):
        return nums
    if not nums:
        return 0.0
    return float(min(nums))


def _fn_max(ev, args):
    nums = ev._collect_numbers(args, 'MAX')
    if is_error(nums):
        return nums
    if not nums:
        return 0.0
    return float(max(nums))


def _fn_count(ev, args):
    nums = ev._collect_numbers(args, 'COUNT')
    if is_error(nums):
        return nums
    return float(len(nums))


def _fn_if(ev, args):
    if len(args) < 2 or len(args) > 3:
        return VALUE
    cond_node = args[0]
    if cond_node[0] == 'range':
        return VALUE
    cond = ev.eval(cond_node)
    if is_error(cond):
        return cond
    if isinstance(cond, str):  # text condition
        return VALUE
    if cond is None:
        truthy = False
    elif isinstance(cond, bool):
        truthy = cond
    else:
        truthy = cond != 0.0
    if truthy:
        branch = args[1]
    else:
        branch = args[2] if len(args) == 3 else None
    if branch is None:
        return False
    if branch[0] == 'range':
        return VALUE
    v = ev.eval(branch)
    return v


def _fn_concat(ev, args):
    parts = []
    for a in args:
        if a[0] == 'range':
            _, r1, c1, r2, c2 = a
            if r1 > 9999 or r2 > 9999 or c1 > 702 or c2 > 702:
                return REF
            for r in range(r1, r2 + 1):
                for c in range(c1, c2 + 1):
                    v = ev.sheet.get_value(r, c)
                    if is_error(v):
                        return v
                    parts.append(to_text(v))
        else:
            v = ev.eval(a)
            if is_error(v):
                return v
            parts.append(to_text(v))
    return ''.join(parts)


def _fn_len(ev, args):
    if len(args) != 1:
        return VALUE
    a = args[0]
    if a[0] == 'range':
        return VALUE
    v = ev.eval(a)
    if is_error(v):
        return v
    return float(len(to_text(v)))


def _fn_abs(ev, args):
    if len(args) != 1:
        return VALUE
    a = args[0]
    if a[0] == 'range':
        return VALUE
    v = ev.eval(a)
    if is_error(v):
        return v
    x = to_number(v)
    if is_error(x):
        return x
    r = abs(x)
    if not math.isfinite(r):
        return VALUE
    return float(r)


def _fn_round(ev, args):
    if len(args) < 1 or len(args) > 2:
        return VALUE
    for a in args:
        if a[0] == 'range':
            return VALUE
    v = ev.eval(args[0])
    if is_error(v):
        return v
    x = to_number(v)
    if is_error(x):
        return x
    if len(args) == 2:
        dv = ev.eval(args[1])
        if is_error(dv):
            return dv
        dnum = to_number(dv)
        if is_error(dnum):
            return dnum
        digits = int(dnum)  # truncated toward zero
    else:
        digits = 0
    if not math.isfinite(x):
        return VALUE
    try:
        r = _round_half_away(x, digits)
    except (OverflowError, ValueError):
        return VALUE
    if not math.isfinite(r):
        return VALUE
    return float(r)


_FUNCS = {
    'SUM': _fn_sum,
    'AVERAGE': _fn_average,
    'MIN': _fn_min,
    'MAX': _fn_max,
    'COUNT': _fn_count,
    'IF': _fn_if,
    'CONCAT': _fn_concat,
    'LEN': _fn_len,
    'ABS': _fn_abs,
    'ROUND': _fn_round,
}


# ---------------------------------------------------------------------------
# Formula static analysis: referenced cells/ranges
# ---------------------------------------------------------------------------

def analyze_formula(ast):
    """Return (refs, ranges, has_parse_error) for a formula AST.

    refs: set of (row, col) single-cell references.
    ranges: list of (r1, c1, r2, c2).
    """
    refs = set()
    ranges = []

    def walk(node):
        tag = node[0]
        if tag == 'ref':
            _, r, c, _canon = node
            refs.add((r, c))
        elif tag == 'range':
            _, r1, c1, r2, c2 = node
            ranges.append((r1, c1, r2, c2))
        elif tag == 'bin':
            walk(node[2])
            walk(node[3])
        elif tag == 'un':
            walk(node[2])
        elif tag == 'call':
            for a in node[2]:
                walk(a)

    walk(ast)
    return refs, ranges

class _Cell:
    __slots__ = ('raw', 'is_formula', 'ast', 'value')

    def __init__(self):
        self.raw = None
        self.is_formula = False
        self.ast = None
        self.value = None


class Sheet:
    def __init__(self):
        self._cells = {}          # (row, col) -> _Cell
        self._undo_stack = []     # list of (rc, prev_raw) tuples

    # -- basic API ----------------------------------------------------------

    def set(self, ref, raw):
        if not isinstance(raw, str):
            raise TypeError('raw must be a string')
        rc = parse_ref(ref)
        if rc is None:
            raise ValueError('invalid cell reference: %r' % (ref,))
        row, col = rc
        prev_raw = None
        cell = self._cells.get(rc)
        if cell is not None:
            prev_raw = cell.raw
        if raw == '':
            # clear the cell
            if cell is not None:
                cell.raw = None
                cell.is_formula = False
                cell.ast = None
                cell.value = None
                if prev_raw is None:
                    # cell was already empty; nothing changed
                    self._cells.pop(rc, None)
                    self._undo_stack.append((rc, None))
                    self._recompute_all()
                    return
        else:
            if cell is None:
                cell = _Cell()
                self._cells[rc] = cell
            cell.raw = raw
            if raw.startswith('='):
                cell.is_formula = True
                cell.ast = parse_formula(raw[1:])
            else:
                cell.is_formula = False
                cell.ast = None
        self._undo_stack.append((rc, prev_raw))
        self._recompute_all()

    def get_raw(self, ref):
        rc = parse_ref(ref)
        if rc is None:
            raise ValueError('invalid cell reference: %r' % (ref,))
        cell = self._cells.get(rc)
        if cell is None or cell.raw is None:
            return None
        return cell.raw

    def get(self, ref):
        rc = parse_ref(ref)
        if rc is None:
            raise ValueError('invalid cell reference: %r' % (ref,))
        return self.get_value(rc[0], rc[1])

    def undo(self):
        if not self._undo_stack:
            return False
        rc, prev_raw = self._undo_stack.pop()
        cell = self._cells.get(rc)
        if prev_raw is None:
            # restore to empty
            if cell is not None:
                cell.raw = None
                cell.is_formula = False
                cell.ast = None
                cell.value = None
            self._cells.pop(rc, None)
        else:
            if cell is None:
                cell = _Cell()
                self._cells[rc] = cell
            cell.raw = prev_raw
            if prev_raw.startswith('='):
                cell.is_formula = True
                cell.ast = parse_formula(prev_raw[1:])
            else:
                cell.is_formula = False
                cell.ast = None
        self._recompute_all()
        return True

    # -- evaluation ---------------------------------------------------------

    def get_value(self, row, col):
        cell = self._cells.get((row, col))
        if cell is None or cell.raw is None:
            return None
        if not cell.is_formula:
            # literal: number or text
            return self._literal_value(cell.raw)
        return cell.value

    @staticmethod
    def _literal_value(raw):
        text = raw.strip()
        if text == '':
            return raw  # whitespace-only literal text (kept verbatim)
        try:
            f = float(text)
        except ValueError:
            return raw
        if math.isfinite(f):
            return f
        return raw

    def _recompute_all(self):
        """Full recompute: build dependency graph, detect cycles, evaluate."""
        # Collect formula cells.
        formulas = {}
        for rc, cell in self._cells.items():
            if cell.raw is None:
                continue
            if cell.is_formula:
                if cell.ast is None:
                    cell.value = PARSE
                else:
                    formulas[rc] = cell
            else:
                cell.value = self._literal_value(cell.raw)

        # Build direct dependency edges (dep -> set of dependents).
        direct_deps = {}   # rc -> set of formula rc's that directly depend on rc
        dep_sources = {}   # formula rc -> set of rc's it reads
        for rc, cell in formulas.items():
            refs, ranges = analyze_formula(cell.ast)
            sources = set(refs)
            for (r1, c1, r2, c2) in ranges:
                # cap the enumeration to avoid huge ranges blowing up memory
                if (r2 - r1 + 1) * (c2 - c1 + 1) <= 20000:
                    for r in range(r1, r2 + 1):
                        for c in range(c1, c2 + 1):
                            sources.add((r, c))
                else:
                    sources.add(('RANGE', r1, c1, r2, c2))
            dep_sources[rc] = sources
            for s in sources:
                direct_deps.setdefault(s, set()).add(rc)

        # Detect cycles among formula cells using iterative DFS.
        cyclic = set()
        WHITE, GRAY, BLACK = 0, 1, 2
        color = {rc: WHITE for rc in formulas}

        for start in formulas:
            if color.get(start, BLACK) != WHITE:
                continue
            stack = [(start, iter(dep_sources.get(start, ())))]
            color[start] = GRAY
            path = [start]
            on_path = {start}
            while stack:
                node, it = stack[-1]
                advanced = False
                for nxt in it:
                    if nxt not in formulas:
                        continue
                    c = color.get(nxt, BLACK)
                    if c == WHITE:
                        color[nxt] = GRAY
                        stack.append((nxt, iter(dep_sources.get(nxt, ()))))
                        path.append(nxt)
                        on_path.add(nxt)
                        advanced = True
                        break
                    elif c == GRAY and nxt in on_path:
                        # found a cycle: mark everything from nxt onward
                        idx = path.index(nxt)
                        for rc in path[idx:]:
                            cyclic.add(rc)
                        # continue scanning this node's remaining edges
                        continue
                if not advanced:
                    stack.pop()
                    color[node] = BLACK
                    on_path.discard(node)
                    path.pop()

        # Mark cyclic cells.
        for rc in cyclic:
            cell = formulas[rc]
            cell.value = CYCLE

        # Topologically evaluate non-cyclic formula cells (iterative).
        remaining = {rc for rc in formulas if rc not in cyclic}
        # in-degree: number of formula-cell sources (within remaining)
        indeg = {}
        dependents_map = {}
        for rc in remaining:
            srcs = dep_sources[rc]
            cnt = 0
            for s in srcs:
                if s in remaining:
                    cnt += 1
                    dependents_map.setdefault(s, set()).add(rc)
            indeg[rc] = cnt
        ready = [rc for rc in remaining if indeg[rc] == 0]
        order = []
        while ready:
            rc = ready.pop()
            order.append(rc)
            for dep in dependents_map.get(rc, ()):
                indeg[dep] -= 1
                if indeg[dep] == 0:
                    ready.append(dep)

        for rc in order:
            cell = formulas[rc]
            ev = Evaluator(self, rc[0], rc[1])
            cell.value = ev.eval(cell.ast)

        # Any remaining non-cyclic formula (shouldn't happen, but be safe):
        for rc in remaining:
            cell = formulas[rc]
            if cell.value is None:
                ev = Evaluator(self, rc[0], rc[1])
                cell.value = ev.eval(cell.ast)

    # -- dependents ----------------------------------------------------------

    def dependents(self, ref):
        rc = parse_ref(ref)
        if rc is None:
            raise ValueError('invalid cell reference: %r' % (ref,))
        # Build direct dependency graph over formula cells.
        direct = {}  # source rc -> set of dependent formula rc's
        for frc, cell in self._cells.items():
            if cell.raw is None or not cell.is_formula or cell.ast is None:
                continue
            refs, ranges = analyze_formula(cell.ast)
            sources = set(refs)
            for (r1, c1, r2, c2) in ranges:
                if (r2 - r1 + 1) * (c2 - c1 + 1) <= 20000:
                    for r in range(r1, r2 + 1):
                        for c in range(c1, c2 + 1):
                            sources.add((r, c))
                else:
                    sources.add(('RANGE', r1, c1, r2, c2))
            for s in sources:
                direct.setdefault(s, set()).add(frc)

        # BFS over dependents.
        result = set()
        frontier = [rc]
        seen = {rc}
        while frontier:
            cur = frontier.pop()
            for dep in direct.get(cur, ()):
                if dep not in seen:
                    seen.add(dep)
                    result.add(dep)
                    frontier.append(dep)
        return sorted(result, key=lambda t: (t[0], t[1]))

    # -- CSV ------------------------------------------------------------------

    def to_csv(self):
        if not self._cells:
            return ''
        max_row = 0
        max_col = 0
        for (r, c), cell in self._cells.items():
            if cell.raw is not None:
                if r > max_row:
                    max_row = r
                if c > max_col:
                    max_col = c
        if max_row == 0 or max_col == 0:
            return ''
        lines = []
        for r in range(1, max_row + 1):
            fields = []
            for c in range(1, max_col + 1):
                v = self.get_value(r, c)
                fields.append(_csv_field(to_text(v)))
            lines.append(','.join(fields))
        return '\n'.join(lines)


def _csv_field(text):
    if any(ch in text for ch in (',', '"', '\n')):
        return '"' + text.replace('"', '""') + '"'
    return text
