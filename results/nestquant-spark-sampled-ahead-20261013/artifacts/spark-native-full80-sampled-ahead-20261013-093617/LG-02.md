"""Data models for the warehouse inventory system."""
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class Product:
    sku: str
    name: str
    unit_cost: float
    reorder_point: int

    def to_dict(self) -> dict:
        return {
            "sku": self.sku,
            "name": self.name,
            "unit_cost": self.unit_cost,
            "reorder_point": self.reorder_point,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "Product":
        return cls(
            sku=d["sku"],
            name=d["name"],
            unit_cost=float(d["unit_cost"]),
            reorder_point=int(d["reorder_point"]),
        )


@dataclass
class Location:
    code: str
    capacity: int

    def to_dict(self) -> dict:
        return {
            "code": self.code,
            "capacity": self.capacity,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "Location":
        return cls(code=d["code"], capacity=int(d["capacity"]))


@dataclass
class StockLevel:
    sku: str
    location_code: str
    quantity: int

    def to_dict(self) -> dict:
        return {
            "sku": self.sku,
            "location_code": self.location_code,
            "quantity": self.quantity,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "StockLevel":
        return cls(
            sku=d["sku"],
            location_code=d["location_code"],
            quantity=int(d["quantity"]),
        )


@dataclass
class Movement:
    kind: str          # 'receive' | 'ship' | 'transfer' | 'adjust'
    sku: str
    qty: int
    src: Optional[str]
    dst: Optional[str]
    ts: int
    ref: str

    def to_dict(self) -> dict:
        return {
            "kind": self.kind,
            "sku": self.sku,
            "qty": self.qty,
            "src": self.src,
            "dst": self.dst,
            "ts": self.ts,
            "ref": self.ref,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "Movement":
        return cls(
            kind=d["kind"],
            sku=d["sku"],
            qty=int(d["qty"]),
            src=d["src"],
            dst=d["dst"],
            ts=int(d["ts"]),
            ref=d["ref"],
        )
