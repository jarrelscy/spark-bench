from __future__ import annotations

import json
import sys
from fractions import Fraction


# ---------------- Money helpers ----------------

def _parse_money(s: str) -> int:
    """Parse a decimal string like '19.99' or '0.00' into integer cents.

    Exactly two decimals are expected. No float math is used.
    """
    if not isinstance(s, str):
        raise ValueError("money must be a decimal string")
    s = s.strip()
    neg = False
    body = s
    if body.startswith("-"):
        neg = True
        body = body[1:]
    elif body.startswith("+"):
        body = body[1:]
    if "." not in body:
        whole = body
        frac = "00"
    else:
        whole, frac = body.split(".", 1)
        if len(frac) != 2:
            raise ValueError("money string must have exactly two decimals")
    if whole == "":
        whole = "0"
    if not whole.isdigit() or not frac.isdigit():
        raise ValueError("money string is not a valid decimal")
    cents = int(whole) * 100 + int(frac)
    return -cents if neg else cents


def _round_half_up(x: Fraction) -> int:
    """Round a Fraction to the nearest integer, halves round up (toward +inf)."""
    # floor(x + 1/2) gives round-half-up for non-negative values and also
    # behaves as "half away from zero" style upward rounding generally;
    # all quantities here are non-negative.
    return (x + Fraction(1, 2)).__floor__()


# ---------------- Order computation ----------------

def _compute_order(order: dict):
    """Compute subtotal, discount, total and per-line allocations.

    Returns (subtotal, discount, total, lines_info) where lines_info is a
    list of dicts with keys: sku, qty, amount, line_discount, unit_net
    (unit_net is an exact Fraction).
    """
    lines = order.get("lines", [])
    subtotal = 0
    line_amounts = []
    for ln in lines:
        qty = ln["qty"]
        price = _parse_money(ln["price"])
        amount = qty * price
        line_amounts.append(amount)
        subtotal += amount

    coupon = order.get("coupon")
    discount = 0
    if coupon is not None:
        percent = coupon["percent"]
        cap = _parse_money(coupon["cap"])
        raw = Fraction(subtotal * percent, 100)
        discount = _round_half_up(raw)
        if discount > cap:
            discount = cap
        if discount < 0:
            discount = 0
        if discount > subtotal:
            discount = subtotal

    total = subtotal - discount

    # Allocate discount across lines proportionally to line amount.
    shares = []
    if subtotal == 0 or len(lines) == 0:
        shares = [0] * len(lines)
    else:
        allocated = 0
        n = len(lines)
        for i in range(n - 1):
            share = _round_half_up(Fraction(discount * line_amounts[i], subtotal))
            shares.append(share)
            allocated += share
        shares.append(discount - allocated)  # last line takes the remainder

    lines_info = []
    for i, ln in enumerate(lines):
        amount = line_amounts[i]
        ld = shares[i]
        if ld > amount:
            ld = amount
        unit_net = Fraction(amount - ld, ln["qty"])
        lines_info.append({
            "sku": ln["sku"],
            "qty": ln["qty"],
            "amount": amount,
            "line_discount": ld,
            "unit_net": unit_net,
        })

    return subtotal, discount, total, lines_info


# ---------------- Ledger engine ----------------

_VALID_STATES = {
    "NEW", "AWAITING_PAYMENT", "PARTIALLY_PAID", "PAID",
    "SHIPPED", "DELIVERED", "CANCELLED", "CLOSED",
}


class _Ledger:
    def __init__(self, order: dict):
        self.order_id = order["id"]
        self.state = "NEW"
        self.subtotal, self.discount, self.total, self.lines = _compute_order(order)
        self.paid = 0
        self.refunded = 0
        self.delivered_ts = None
        self.returned = {}   # sku -> returned qty so far
        self.line_returned_refunds = [0] * len(self.lines)  # refunds per line
        self.audit = []      # list of audit dicts (applied/rejected)
        self.applied = 0
        self.rejected = 0
        self.duplicates = 0
        self.seen_ids = set()

    # -- helpers --

    def _audit(self, ev: dict, result: str, reason: str = ""):
        entry = {
            "id": ev["id"],
            "type": ev["type"],
            "ts": ev["ts"],
            "result": result,
            "state_after": self.state,
        }
        if reason:
            entry["reason"] = reason
        self.audit.append(entry)

    def _reject(self, ev: dict, reason: str):
        self.rejected += 1
        self._audit(ev, "rejected", reason)

    def _apply(self, ev: dict):
        self.applied += 1
        self._audit(ev, "applied")

    def _line_index_for_sku(self, sku: str):
        """Return index of the first line with this sku, or None."""
        for i, ln in enumerate(self.lines):
            if ln["sku"] == sku:
                return i
        return None

    # -- event dispatch --

    def process(self, events: list):
        ordered = sorted(enumerate(events), key=lambda p: (p[1]["ts"], p[0]))
        for _, ev in ordered:
            eid = ev["id"]
            if eid in self.seen_ids:
                self.duplicates += 1
                continue
            self.seen_ids.add(eid)
            self._handle(ev)

    def _handle(self, ev: dict):
        etype = ev.get("type")
        handler = {
            "place": self._ev_place,
            "pay": self._ev_pay,
            "ship": self._ev_ship,
            "deliver": self._ev_deliver,
            "cancel": self._ev_cancel,
            "return": self._ev_return,
            "close": self._ev_close,
        }.get(etype)
        if handler is None:
            self._reject(ev, "unknown event type")
            return
        handler(ev)

    def _ev_place(self, ev: dict):
        if self.state != "NEW":
            self._reject(ev, "place only allowed in NEW")
            return
        if len(self.lines) == 0:
            self._reject(ev, "order has no lines")
            return
        self.state = "AWAITING_PAYMENT"
        if self.total == 0:
            self.state = "PAID"
        self._apply(ev)

    def _ev_pay(self, ev: dict):
        if self.state not in ("AWAITING_PAYMENT", "PARTIALLY_PAID"):
            self._reject(ev, "pay not allowed in state " + self.state)
            return
        if self.total == 0:
            self._reject(ev, "pay rejected: total is 0")
            return
        amount = _parse_money(ev["amount"])
        if amount <= 0:
            self._reject(ev, "pay amount must be > 0")
            return
        if self.paid + amount > self.total:
            self._reject(ev, "pay would exceed total")
            return
        self.paid += amount
        if self.paid == self.total:
            self.state = "PAID"
        else:
            self.state = "PARTIALLY_PAID"
        self._apply(ev)

    def _ev_ship(self, ev: dict):
        if self.state != "PAID":
            self._reject(ev, "ship only allowed in PAID")
            return
        self.state = "SHIPPED"
        self._apply(ev)

    def _ev_deliver(self, ev: dict):
        if self.state != "SHIPPED":
            self._reject(ev, "deliver only allowed in SHIPPED")
            return
        self.state = "DELIVERED"
        self.delivered_ts = ev["ts"]
        self._apply(ev)

    def _ev_cancel(self, ev: dict):
        if self.state in ("NEW", "AWAITING_PAYMENT"):
            self.state = "CANCELLED"
            self._apply(ev)
            return
        if self.state in ("PARTIALLY_PAID", "PAID"):
            refund = self.paid - self.refunded
            self.refunded += refund
            self.state = "CANCELLED"
            self._apply(ev)
            return
        self._reject(ev, "cancel not allowed in state " + self.state)

    def _ev_return(self, ev: dict):
        if self.state != "DELIVERED":
            self._reject(ev, "return only allowed in DELIVERED")
            return
        if self.delivered_ts is None or (ev["ts"] - self.delivered_ts) > 30:
            self._reject(ev, "return window expired")
            return
        sku = ev.get("sku")
        qty = ev.get("qty")
        if not isinstance(qty, int) or qty < 1:
            self._reject(ev, "return qty must be >= 1")
            return
        li = self._line_index_for_sku(sku)
        if li is None:
            self._reject(ev, "unknown sku")
            return
        line = self.lines[li]
        already = self.returned.get(sku, 0)
        if already + qty > line["qty"]:
            self._reject(ev, "return exceeds ordered qty")
            return
        # Refund for this return: round(qty * unit_net), capped so the line's
        # total refunds never exceed the line's net amount. When the line
        # becomes fully returned, refund exactly the remaining net amount.
        line_net = line["amount"] - line["line_discount"]
        raw_refund = _round_half_up(Fraction(qty) * line["unit_net"])
        if already + qty == line["qty"]:
            refund = line_net - self.line_returned_refunds[li]
        else:
            refund = raw_refund
            if self.line_returned_refunds[li] + refund > line_net:
                refund = line_net - self.line_returned_refunds[li]
        self.line_returned_refunds[li] += refund
        self.refunded += refund
        self.returned[sku] = already + qty
        # If every unit of every line is returned -> CLOSED.
        all_returned = True
        for ln in self.lines:
            if self.returned.get(ln["sku"], 0) < ln["qty"]:
                all_returned = False
                break
        if all_returned:
            self.state = "CLOSED"
        self._apply(ev)

    def _ev_close(self, ev: dict):
        if self.state != "DELIVERED":
            self._reject(ev, "close only allowed in DELIVERED")
            return
        self.state = "CLOSED"
        self._apply(ev)

    # -- result --

    def result(self) -> dict:
        returned_out = {sku: q for sku, q in self.returned.items() if q > 0}
        return {
            "state": self.state,
            "subtotal": self.subtotal,
            "discount": self.discount,
            "total": self.total,
            "paid": self.paid,
            "refunded": self.refunded,
            "applied": self.applied,
            "rejected": self.rejected,
            "duplicates": self.duplicates,
            "returned": returned_out,
            "audit_len": len(self.audit),
            "replay_ok": False,
        }


def _replay(order: dict, audit: list) -> bool:
    """Re-apply only the applied audit entries, in audit order, on a fresh
    order. Returns True if the second run ends in the same state with the
    same paid and refunded totals and rejects nothing."""
    ev_by_id = {ev["id"]: ev for ev in []}  # placeholder, replaced below
    # We need the original event dicts; the audit entries carry id/type/ts,
    # but the caller passes the audit list which contains the event dicts
    # merged in. To be safe, we reconstruct events from audit entries.
    replay_events = []
    for entry in audit:
        if entry.get("result") != "applied":
            continue
        ev = entry.get("_event")
        if ev is None:
            # Fallback: reconstruct minimal event from audit entry.
            ev = {"id": entry["id"], "type": entry["type"], "ts": entry["ts"]}
            for k, v in entry.items():
                if k not in ("id", "type", "ts", "result", "state_after", "reason"):
                    ev[k] = v
        replay_events.append(ev)

    fresh = _Ledger(order)
    fresh.process(replay_events)
    return (
        fresh.state == None or True  # placeholder never used
    )


def run_trace(order: dict, events: list) -> dict:
    ledger = _Ledger(order)
    ledger.process(events)
    res = ledger.result()

    # Build replay: re-apply only applied audit entries, in audit order,
    # using the same event dicts.
    applied_events = []
    audit_by_id = {}
    for entry in ledger.audit:
        audit_by_id[entry["id"]] = entry
    ev_by_id = {}
    for ev in events:
        if ev["id"] not in ev_by_id:
            ev_by_id[ev["id"]] = ev
    for entry in ledger.audit:
        if entry["result"] != "applied":
            continue
        ev = ev_by_id.get(entry["id"])
        if ev is None:
            ev = {"id": entry["id"], "type": entry["type"], "ts": entry["ts"]}
        applied_events.append(ev)

    fresh = _Ledger(order)
    fresh.process(applied_events)

    res["replay_ok"] = (
        fresh.rejected == 0
        and fresh.state == ledger.state
        and fresh.paid == ledger.paid
        and fresh.refunded == ledger.refunded
    )
    return res
