"""Data models for the warehouse inventory system."""
from dataclasses import dataclass, field, fields, asdict
from typing import Optional


@dataclass
class Product:
    sku: str
    name: str
    unit_cost: float
    reorder_point: int

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "Product":
        return cls(
            sku=str(d["sku"]),
            name=str(d["name"]),
            unit_cost=float(d["unit_cost"]),
            reorder_point=int(d["reorder_point"]),
        )


@dataclass
class Location:
    code: str
    capacity: int

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "Location":
        return cls(code=str(d["code"]), capacity=int(d["capacity"]))


@dataclass
class StockLevel:
    sku: str
    location_code: str
    quantity: int

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "StockLevel":
        return cls(
            sku=str(d["sku"]),
            location_code=str(d["location_code"]),
            quantity=int(d["quantity"]),
        )


@dataclass
class Movement:
    kind: str          # 'receive' | 'ship' | 'transfer' | 'adjust'
    sku: str
    qty: int
    src: Optional[str]
    dst: Optional[str]
    ts: int
    ref: str

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "Movement":
        return cls(
            kind=str(d["kind"]),
            sku=str(d["sku"]),
            qty=int(d["qty"]),
            src=(None if d.get("src") is None else str(d["src"])),
            dst=(None if d.get("dst") is None else str(d["dst"])),
            ts=int(d["ts"]),
            ref=str(d["ref"]),
        )
