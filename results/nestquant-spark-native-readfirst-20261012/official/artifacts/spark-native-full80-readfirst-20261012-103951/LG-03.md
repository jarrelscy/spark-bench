"""A spreadsheet formula engine (standard library only, Python 3.11)."""

from __future__ import annotations

import math
import re
from decimal import Decimal, ROUND_HALF_UP, localcontext

__all__ = ["Sheet"]

MAX_ROWS = 9999
MAX_COLS = 26 * 26  # A..ZZ

_ERROR_CODES = {
    "#DIV/0!",
    "#VALUE!",
    "#REF!",
    "#NAME?",
    "#PARSE!",
    "#CYCLE!",
}

_REF_RE = re.compile(r"^[A-Za-z]{1,2}[0-9]{1,4}$")


def _col_to_num(letters: str) -> int:
    """Excel-style column letters (A..Z, AA..ZZ) to 1-based index."""
    n = 0
    for ch in letters:
        n = n * 26 + (ord(ch) - ord("A") + 1)
    return n


def _num_to_col(n: int) -> str:
    """1-based Excel-style column index to letters (A..Z, AA..ZZ)."""
    if n <= 26:
        return chr(ord("A") + n - 1)
    n -= 1
    return _num_to_col(n // 26) + chr(ord("A") + n % 26)


def _parse_ref(ref: str) -> tuple[int, int]:
    """Parse a cell reference string -> (row, col) 1-based, or raise ValueError."""
    if not isinstance(ref, str):
        raise ValueError("invalid reference")
    m = _REF_RE.match(ref)
    if not m:
        raise ValueError("invalid reference")
    letters = ref[: len(ref) - len(ref.lstrip("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"))].upper()
    digits = ref[len(letters):]
    col = _col_to_num(letters)
    row = int(digits)
    if row < 1 or row > MAX_ROWS or col < 1 or col > MAX_COLS:
        raise ValueError("invalid reference")
    return (row, col)


def _canonical_ref(ref: str) -> str:
    row, col = _parse_ref(ref)
    return _num_to_col(col) + str(row)


class _Cell:
    __slots__ = ("raw", "is_formula", "value", "state")

    def __init__(self) -> None:
        self.raw: str | None = None
        self.is_formula: bool = False
        self.value: object = None
        self.state: int = 0  # 0 = clean, 1 = visiting, 2 = cyclic


# ---------------------------------------------------------------------------
# Formula AST nodes
# ---------------------------------------------------------------------------


class _Node:
    __slots__ = ()


class _Num(_Node):
    __slots__ = ("value",)

    def __init__(self, value: float) -> None:
        self.value = value


class _Str(_Node):
    __slots__ = ("value",)

    def __init__(self, value: str) -> None:
        self.value = value


class _Bool(_Node):
    __slots__ = ("value",)

    def __init__(self, value: bool) -> None:
        self.value = value


class _Ref(_Node):
    __slots__ = ("row", "col")

    def __init__(self, row: int, col: int) -> None:
        self.row = row
        self.col = col


class _Range(_Node):
    __slots__ = ("r1", "c1", "r2", "c2")

    def __init__(self, r1: int, c1: int, r2: int, c2: int) -> None:
        self.r1 = r1
        self.c1 = c1
        self.r2 = r2
        self.c2 = c2


class _Bin(_Node):
    __slots__ = ("op", "left", "right")

    def __init__(self, op: str, left: _Node, right: _Node) -> None:
        self.op = op
        self.left = left
        self.right = right


class _Un(_Node):
    __slots__ = ("op", "operand")

    def __init__(self, op: str, operand: _Node) -> None:
        self.op = op
        self.operand = operand


class _Call(_Node):
    __slots__ = ("name", "args")

    def __init__(self, name: str, args: list[_Node]) -> None:
        self.name = name
        self.args = args


class _Bad(_Node):
    """A node that always evaluates to a fixed error (e.g. out-of-bounds ref)."""
    __slots__ = ("err",)

    def __init__(self, err: str) -> None:
        self.err = err


class _ParseError(Exception):
    pass


# ---------------------------------------------------------------------------
# Tokenizer
# ---------------------------------------------------------------------------

_TOKEN_NUM = re.compile(r"[0-9]*\.?[0-9]+(?:[eE][+-]?[0-9]+)?")
_TOKEN_IDENT = re.compile(r"[A-Za-z][A-Za-z0-9_]*")
_TOKEN_REF = re.compile(r"[A-Za-z]{1,2}[0-9]{1,4}")
_TOKEN_CMP = re.compile(r"<=|>=|<>|[=<>]")
_TOKEN_WS = re.compile(r"[ \t\r\n]+")


def _tokenize(src: str) -> list[tuple[str, str]]:
    """Return a list of (kind, text) tokens. Kinds: num, str, ident, op, lparen,
    rparen, comma, colon, range_end."""
    tokens: list[tuple[str, str]] = []
    i = 0
    n = len(src)
    while i < n:
        ch = src[i]
        if ch in " \t\r\n":
            i += 1
            continue
        if ch == '"':
            j = i + 1
            buf = []
            while True:
                if j >= n:
                    raise _ParseError("unterminated string")
                if src[j] == '"':
                    if j + 1 < n and src[j + 1] == '"':
                        buf.append('"')
                        j += 2
                    else:
                        j += 1
                        break
                else:
                    buf.append(src[j])
                    j += 1
            tokens.append(("str", "".join(buf)))
            i = j
            continue
        if ch.isdigit() or (ch == "." and i + 1 < n and src[i + 1].isdigit()):
            m = _TOKEN_NUM.match(src, i)
            tokens.append(("num", m.group(0)))
            i = m.end()
            continue
        if ch.isalpha():
            m = _TOKEN_IDENT.match(src, i)
            text = m.group(0)
            rest = src[m.end():].lstrip(" \t\r\n")
            if rest.startswith(":"):
                # Possibly a range: ident must be reference-shaped; the colon
                # belongs to the range operator.
                if _TOKEN_REF.fullmatch(text):
                    tokens.append(("ident", text))
                    i = m.end()
                    continue
                raise _ParseError("bad range")
            tokens.append(("ident", text))
            i = m.end()
            continue
        if ch == "(":
            tokens.append(("lparen", ch))
            i += 1
            continue
        if ch == ")":
            tokens.append(("rparen", ch))
            i += 1
            continue
        if ch == ",":
            tokens.append(("comma", ch))
            i += 1
            continue
        if ch == ":":
            tokens.append(("colon", ch))
            i += 1
            continue
        if ch in "+-*/^&":
            tokens.append(("op", ch))
            i += 1
            continue
        m = _TOKEN_CMP.match(src, i)
        if m:
            tokens.append(("op", m.group(0)))
            i = m.end()
            continue
        raise _ParseError("unexpected character")
    return tokens


# ---------------------------------------------------------------------------
# Parser (recursive descent over tokens)
# ---------------------------------------------------------------------------


class _Parser:
    def __init__(self, tokens: list[tuple[str, str]]) -> None:
        self.tokens = tokens
        self.pos = 0

    def _peek(self) -> tuple[str, str] | None:
        if self.pos < len(self.tokens):
            return self.tokens[self.pos]
        return None

    def _next(self) -> tuple[str, str]:
        tok = self._peek()
        if tok is None:
            raise _ParseError("unexpected end of formula")
        self.pos += 1
        return tok

    def _expect(self, kind: str) -> tuple[str, str]:
        tok = self._peek()
        if tok is None or tok[0] != kind:
            raise _ParseError("expected " + kind)
        self.pos += 1
        return tok

    def parse(self) -> _Node:
        node = self.parse_cmp()
        if self._peek() is not None:
            raise _ParseError("trailing tokens")
        return node

    def parse_cmp(self) -> _Node:
        left = self.parse_concat()
        while True:
            tok = self._peek()
            if tok is not None and tok[0] == "op" and tok[1] in ("=", "<>", "<", ">", "<=", ">="):
                self._next()
                right = self.parse_concat()
                left = _Bin(tok[1], left, right)
            else:
                return left

    def parse_concat(self) -> _Node:
        left = self.parse_add()
        while True:
            tok = self._peek()
            if tok is not None and tok[0] == "op" and tok[1] == "&":
                self._next()
                right = self.parse_add()
                left = _Bin("&", left, right)
            else:
                return left

    def parse_add(self) -> _Node:
        left = self.parse_mul()
        while True:
            tok = self._peek()
            if tok is not None and tok[0] == "op" and tok[1] in ("+", "-"):
                self._next()
                right = self.parse_mul()
                left = _Bin(tok[1], left, right)
            else:
                return left

    def parse_mul(self) -> _Node:
        left = self.parse_unary()
        while True:
            tok = self._peek()
            if tok is not None and tok[0] == "op" and tok[1] in ("*", "/"):
                self._next()
                right = self.parse_unary()
                left = _Bin(tok[1], left, right)
            else:
                return left

    def parse_unary(self) -> _Node:
        tok = self._peek()
        if tok is not None and tok[0] == "op" and tok[1] in ("+", "-"):
            self._next()
            operand = self.parse_unary()
            return _Un(tok[1], operand)
        return self.parse_power()

    def parse_power(self) -> _Node:
        base = self.parse_primary()
        tok = self._peek()
        if tok is not None and tok[0] == "op" and tok[1] == "^":
            self._next()
            exponent = self.parse_unary()  # right-associative, allows unary after ^
            return _Bin("^", base, exponent)
        return base

    def parse_primary(self) -> _Node:
        tok = self._next()
        kind, text = tok
        if kind == "num":
            try:
                value = float(text)
            except ValueError:
                raise _ParseError("bad number")
            return _Num(value)
        if kind == "str":
            return _Str(text)
        if kind == "lparen":
            node = self.parse_cmp()
            self._expect("rparen")
            return node
        if kind == "ident":
            nxt = self._peek()
            if nxt is not None and nxt[0] == "lparen":
                self._next()
                args: list[_Node] = []
                after = self._peek()
                if after is not None and after[0] == "rparen":
                    self._next()
                else:
                    while True:
                        args.append(self.parse_arg())
                        tok2 = self._next()
                        if tok2[0] == "comma":
                            continue
                        if tok2[0] == "rparen":
                            break
                        raise _ParseError("expected , or )")
                return self._finish_call(text, args)
            # TRUE / FALSE
            up = text.upper()
            if up == "TRUE":
                return _Bool(True)
            if up == "FALSE":
                return _Bool(False)
            # reference-shaped?
            if _TOKEN_REF.fullmatch(text):
                row, col = self._ref_rc(text)
                if row < 1 or row > MAX_ROWS or col < 1 or col > MAX_COLS:
                    return _Bad("#REF!")
                # range?
                nxt = self._peek()
                if nxt is not None and nxt[0] == "colon":
                    self._next()
                    tok3 = self._next()
                    if tok3[0] != "ident" or not _TOKEN_REF.fullmatch(tok3[1]):
                        raise _ParseError("bad range")
                    r2, c2 = self._ref_rc(tok3[1])
                    return _Range(row, col, r2, c2)
                return _Ref(row, col)
            # out-of-bounds reference-shaped token (e.g. A10000, AAA1)?
            m = re.fullmatch(r"[A-Za-z]{1,6}[0-9]{1,7}", text)
            if m and re.fullmatch(r"[A-Za-z]+", text[: m.start() + len(text) - len(text.lstrip("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"))] if False else ""):
                pass  # unreachable; kept for clarity
            letters = text
            digits = ""
            for k, ch in enumerate(text):
                if ch.isdigit():
                    letters = text[:k]
                    digits = text[k:]
                    break
            if digits and all(c.isdigit() for c in digits) and letters.isalpha():
                return _Bad("#REF!")
            return _Bad("#NAME?")
        raise _ParseError("unexpected token")

    def parse_arg(self) -> _Node:
        """An argument: expression, or a range (ident ':' ident)."""
        tok = self._peek()
        if tok is not None and tok[0] == "ident" and _TOKEN_REF.fullmatch(tok[1]):
            # Look ahead for ':' (skipping whitespace already done by tokenizer).
            if self.pos + 1 < len(self.tokens) and self.tokens[self.pos + 1][0] == "colon":
                self._next()
                self._next()
                tok3 = self._next()
                if tok3[0] != "ident" or not _TOKEN_REF.fullmatch(tok3[1]):
                    raise _ParseError("bad range")
                r1, c1 = self._ref_rc(tok[1])
                r2, c2 = self._ref_rc(tok3[1])
                return _Range(r1, c1, r2, c2)
        return self.parse_cmp()

    @staticmethod
    def _ref_rc(text: str) -> tuple[int, int]:
        letters = text.rstrip("0123456789")
        digits = text[len(letters):]
        try:
            col = _col_to_num(letters.upper())
            row = int(digits)
        except Exception:
            return (0, 0)
        return (row, col)

    def _finish_call(self, name: str, args: list[_Node]) -> _Node:
        # A range used as a top-level argument is fine (function context);
        # the evaluator decides whether the function accepts ranges.
        return _Call(name.upper(), args)


def _parse_formula(src: str) -> _Node:
    tokens = _tokenize(src)
    if not tokens:
        raise _ParseError("empty formula")
    return _Parser(tokens).parse()

# Evaluation helpers

def _is_err(v: object) -> bool:
    return isinstance(v, str) and v in _ERROR_CODES


def _to_number(v: object) -> float | str:
    """Arithmetic coercion. Returns float or an error string."""
    if isinstance(v, bool):
        return 1.0 if v else 0.0
    if isinstance(v, float):
        return v
    if v is None:
        return 0.0
    if isinstance(v, str):
        return "#VALUE!"
    return "#VALUE!"


def _to_text(v: object) -> str:
    """Text coercion."""
    if v is None:
        return ""
    if isinstance(v, bool):
        return "TRUE" if v else "FALSE"
    if isinstance(v, str):
        return v
    if isinstance(v, float):
        if math.isfinite(v) and v == int(v) and abs(v) < 1e15:
            return str(int(v))
        return repr(v)
    return str(v)


def _compare(a: object, b: object) -> object:
    """Comparison coercion: returns (kind, x, y) or an error string.
    kind is 'num' or 'text'."""
    if _is_err(a):
        return a
    if _is_err(b):
        return b
    if isinstance(a, bool):
        a = 1.0 if a else 0.0
    if isinstance(b, bool):
        b = 1.0 if b else 0.0
    a_empty = a is None
    b_empty = b is None
    if a_empty:
        a = 0.0
    if b_empty:
        b = 0.0
    a_num = isinstance(a, float)
    b_num = isinstance(b, float)
    if a_num and b_num:
        return ("num", a, b)
    if not a_num and not b_num:
        return ("text", _to_text(a).lower(), _to_text(b).lower())
    # mixed: number vs text -> #VALUE!
    return "#VALUE!"


def _cmp_op(kind: str, x: object, y: object, op: str) -> bool:
    if kind == "num":
        if op == "=":
            return x == y  # type: ignore[operator]
        if op == "<>":
            return x != y  # type: ignore[operator]
        if op == "<":
            return x < y  # type: ignore[operator]
        if op == ">":
            return x > y  # type: ignore[operator]
        if op == "<=":
            return x <= y  # type: ignore[operator]
        return x >= y  # type: ignore[operator]
    else:
        if op == "=":
            return x == y
        if op == "<>":
            return x != y
        if op == "<":
            return x < y  # type: ignore[operator]
        if op == ">":
            return x > y  # type: ignore[operator]
        if op == "<=":
            return x <= y  # type: ignore[operator]
        return x >= y  # type: ignore[operator]


def _round_half_away(x: float, digits: int) -> float:
    """Round half away from zero to `digits` decimals, exactly on decimal ties."""
    if not math.isfinite(x):
        return x
    with localcontext() as ctx:
        ctx.prec = 60
        d = Decimal(repr(x)) if not isinstance(x, Decimal) else x
        q = Decimal(1).scaleb(-digits)
        try:
            r = d.quantize(q, rounding=ROUND_HALF_UP)
        except Exception:
            r = d
        return float(r)

class Sheet:
    def __init__(self) -> None:
        self._cells: dict[tuple[int, int], _Cell] = {}
        self._formulas: dict[tuple[int, int], _Node] = {}
        self._undo_stack: list[tuple[tuple[int, int], str | None]] = []

    # -- public API ---------------------------------------------------------

    def set(self, ref: str, raw: str) -> None:
        if not isinstance(raw, str):
            raise TypeError("raw must be a string")
        row, col = _parse_ref(ref)
        key = (row, col)
        cell = self._cells.get(key)
        if cell is None:
            cell = _Cell()
            self._cells[key] = cell
        old_raw = cell.raw
        if raw == "":
            cell.raw = None
            cell.is_formula = False
            self._formulas.pop(key, None)
        else:
            cell.raw = raw
            if raw.startswith("="):
                cell.is_formula = True
                try:
                    ast = _parse_formula(raw[1:])
                    self._formulas[key] = ast
                except _ParseError:
                    self._formulas[key] = _Bad("#PARSE!")
            else:
                cell.is_formula = False
                self._formulas.pop(key, None)
        self._undo_stack.append((key, old_raw))
        self._reevaluate()

    def get_raw(self, ref: str) -> str | None:
        row, col = _parse_ref(ref)
        cell = self._cells.get((row, col))
        if cell is None or cell.raw is None:
            return None
        return cell.raw

    def get(self, ref: str):
        row, col = _parse_ref(ref)
        cell = self._cells.get((row, col))
        if cell is None or cell.raw is None:
            return None
        if cell.is_formula:
            return cell.value
        return self._eval_literal(cell.raw)

    def undo(self) -> bool:
        if not self._undo_stack:
            return False
        key, old_raw = self._undo_stack.pop()
        cell = self._cells.get(key)
        if cell is None:
            cell = _Cell()
            self._cells[key] = cell
        cell.raw = old_raw
        if old_raw is None:
            cell.is_formula = False
            self._formulas.pop(key, None)
        elif old_raw.startswith("="):
            cell.is_formula = True
            try:
                self._formulas[key] = _parse_formula(old_raw[1:])
            except _ParseError:
                self._formulas[key] = _Bad("#PARSE!")
        else:
            cell.is_formula = False
            self._formulas.pop(key, None)
        self._reevaluate()
        return True

    def dependents(self, ref: str) -> list[str]:
        row, col = _parse_ref(ref)
        target = (row, col)
        # Build reverse dependency edges: dep -> set of cells that depend on it.
        rev: dict[tuple[int, int], set[tuple[int, int]]] = {}
        for key, ast in self._formulas.items():
            for dep in self._ast_refs(ast):
                rev.setdefault(dep, set()).add(key)
        # BFS from target over reverse edges.
        seen: set[tuple[int, int]] = {target}
        frontier = [target]
        result: set[tuple[int, int]] = set()
        while frontier:
            cur = frontier.pop()
            for dep_cell in rev.get(cur, ()):
                if dep_cell not in seen:
                    seen.add(dep_cell)
                    result.add(dep_cell)
                    frontier.append(dep_cell)
        return [
            _num_to_col(c) + str(r)
            for (r, c) in sorted(result)
        ]

    def to_csv(self) -> str:
        if not self._cells:
            return ""
        max_row = 0
        max_col = 0
        for (r, c), cell in self._cells.items():
            if cell.raw is not None:
                if r > max_row:
                    max_row = r
                if c > max_col:
                    max_col = c
        lines: list[str] = []
        for r in range(1, max_row + 1):
            fields: list[str] = []
            for c in range(1, max_col + 1):
                cell = self._cells.get((r, c))
                if cell is None or cell.raw is None:
                    fields.append("")
                else:
                    v = self.get(_num_to_col(c) + str(r))
                    fields.append(self._csv_field(_to_text(v)))
            lines.append(",".join(fields))
        return "\n".join(lines)

    # -- internals ----------------------------------------------------------

    @staticmethod
    def _csv_field(s: str) -> str:
        if "," in s or '"' in s or "\n" in s:
            return '"' + s.replace('"', '""') + '"'
        return s

    @staticmethod
    def _eval_literal(raw: str):
        text = raw.strip()
        if text == "":
            return None
        try:
            # Only accept whole-text numbers.
            if re.fullmatch(r"[+-]?([0-9]+\.?[0-9]*|\.[0-9]+)([eE][+-]?[0-9]+)?", text):
                return float(text)
        except ValueError:
            pass
        return raw

    def _ast_refs(self, node: _Node):
        """Yield every (row, col) cell that this AST directly depends on."""
        stack = [node]
        while stack:
            cur = stack.pop()
            if isinstance(cur, _Ref):
                yield (cur.row, cur.col)
            elif isinstance(cur, _Range):
                r1, r2 = min(cur.r1, cur.r2), max(cur.r1, cur.r2)
                c1, c2 = min(cur.c1, cur.c2), max(cur.c1, cur.c2)
                for r in range(r1, r2 + 1):
                    for c in range(c1, c2 + 1):
                        yield (r, c)
            elif isinstance(cur, _Bin):
                stack.append(cur.left)
                stack.append(cur.right)
            elif isinstance(cur, _Un):
                stack.append(cur.operand)
            elif isinstance(cur, _Call):
                stack.extend(cur.args)

    def _reevaluate(self) -> None:
        """Re-evaluate all formula cells, detecting cycles."""
        for cell in self._cells.values():
            cell.state = 0
        order: list[tuple[int, int]] = []
        cyclic: set[tuple[int, int]] = set()
        color: dict[tuple[int, int], int] = {}

        # Iterative DFS to find cyclic cells (nodes on back edges).
        WHITE, GRAY, BLACK = 0, 1, 2
        for start in list(self._formulas.keys()):
            if color.get(start, WHITE) != WHITE:
                continue
            stack: list[tuple[tuple[int, int], list[tuple[int, int]], int]] = [(start, [], 0)]
            path: list[tuple[int, int]] = []
            path_set: set[tuple[int, int]] = set()
            # Simpler approach: explicit stack of (node, child_index).
            stack2: list[list] = []
            color[start] = GRAY
            path.append(start)
            path_set.add(start)
            deps_list = self._formula_deps(start)
            stack2.append([start, deps_list, 0])
            while stack2:
                frame = stack2[-1]
                node, deps, idx = frame
                if idx >= len(deps):
                    color[node] = BLACK
                    stack2.pop()
                    if path and path[-1] == node:
                        path.pop()
                        if path:
                            path_set.discard(node)
                        else:
                            path_set.clear()
                    continue
                frame[2] += 1
                dep = deps[idx]
                if dep not in self._formulas:
                    continue
                state = color.get(dep, WHITE)
                if state == WHITE:
                    color[dep] = GRAY
                    path.append(dep)
                    path_set.add(dep)
                    stack2.append([dep, self._formula_deps(dep), 0])
                elif state == GRAY:
                    # Found a cycle: everything on the path from dep to node.
                    try:
                        pos = path.index(dep)
                    except ValueError:
                        pos = 0
                    for k in range(pos, len(path)):
                        cyclic.add(path[k])
        for key in cyclic:
            cell = self._cells.get(key)
            if cell is not None:
                cell.state = 2
                cell.value = "#CYCLE!"

        # Evaluate in dependency order (topological), iterative.
        evaluated: set[tuple[int, int]] = set()
        for start in list(self._formulas.keys()):
            if start in evaluated or self._cells[start].state == 2:
                evaluated.add(start)
                continue
            # Iterative post-order evaluation.
            stack3: list[list] = []
            visiting: set[tuple[int, int]] = set()
            stack3.append([start, self._formula_deps(start), 0])
            visiting.add(start)
            while stack3:
                frame = stack3[-1]
                node, deps, idx = frame
                if idx >= len(deps):
                    stack3.pop()
                    visiting.discard(node)
                    if node not in evaluated:
                        self._eval_cell(node)
                        evaluated.add(node)
                    continue
                frame[2] += 1
                dep = deps[idx]
                if dep not in self._formulas:
                    continue
                if self._cells[dep].state == 2:
                    continue
                if dep in evaluated:
                    continue
                if dep in visiting:
                    # Shouldn't happen (cycles already marked); treat as cycle.
                    self._cells[dep].value = "#CYCLE!"
                    continue
                visiting.add(dep)
                stack3.append([dep, self._formula_deps(dep), 0])

    def _formula_deps(self, key: tuple[int, int]) -> list[tuple[int, int]]:
        """Direct dependencies of the formula at key (deduplicated)."""
        ast = self._formulas.get(key)
        if ast is None:
            return []
        deps: list[tuple[int, int]] = []
        seen: set[tuple[int, int]] = set()
        for dep in self._ast_refs(ast):
            if dep not in seen:
                seen.add(dep)
                deps.append(dep)
        return deps

    def _eval_cell(self, key: tuple[int, int]) -> None:
        ast = self._formulas.get(key)
        cell = self._cells.get(key)
        if ast is None or cell is None:
            return
        if cell.state == 2:
            cell.value = "#CYCLE!"
            return
        cell.value = self._eval(ast)

    # -- expression evaluation ----------------------------------------------

    def _eval(self, node: _Node):
        if isinstance(node, _Num):
            return node.value
        if isinstance(node, _Str):
            return node.value
        if isinstance(node, _Bool):
            return node.value
        if isinstance(node, _Bad):
            return node.err
        if isinstance(node, _Ref):
            return self._cell_value(node.row, node.col)
        if isinstance(node, _Range):
            # A range outside a function argument: #VALUE!.
            return "#VALUE!"
        if isinstance(node, _Un):
            v = self._eval(node.operand)
            if _is_err(v):
                return v
            n = _to_number(v)
            if _is_err(n):
                return n
            n = float(n)
            if node.op == "-":
                r = -n
            else:
                r = n
            if not math.isfinite(r):
                return "#VALUE!"
            return r
        if isinstance(node, _Bin):
            return self._eval_bin(node)
        if isinstance(node, _Call):
            return self._eval_call(node)
        return "#VALUE!"

    def _cell_value(self, row: int, col: int):
        cell = self._cells.get((row, col))
        if cell is None or cell.raw is None:
            return None
        if cell.is_formula:
            return cell.value
        return self._eval_literal(cell.raw)

    def _range_values(self, rng: _Range):
        """Yield (value, is_error) for each cell in the range, row by row."""
        r1, r2 = min(rng.r1, rng.r2), max(rng.r1, rng.r2)
        c1, c2 = min(rng.c1, rng.c2), max(rng.c1, rng.c2)
        for r in range(r1, r2 + 1):
            for c in range(c1, c2 + 1):
                yield self._cell_value(r, c)

    def _eval_bin(self, node: _Bin):
        op = node.op
        if op in ("=", "<>", "<", ">", "<=", ">="):
            left = self._eval(node.left)
            if _is_err(left):
                return left
            right = self._eval(node.right)
            if _is_err(right):
                return right
            cmp = _compare(left, right)
            if isinstance(cmp, str):
                return cmp
            kind, x, y = cmp
            return _cmp_op(kind, x, y, op)
        left = self._eval(node.left)
        if _is_err(left):
            return left
        right = self._eval(node.right)
        if _is_err(right):
            return right
        if op == "&":
            return _to_text(left) + _to_text(right)
        ln = _to_number(left)
        if _is_err(ln):
            return ln
        rn = _to_number(right)
        if _is_err(rn):
            return rn
        ln = float(ln)
        rn = float(rn)
        if op == "+":
            r = ln + rn
        elif op == "-":
            r = ln - rn
        elif op == "*":
            r = ln * rn
        elif op == "/":
            if rn == 0.0:
                return "#DIV/0!"
            r = ln / rn
        elif op == "^":
            if ln == 0.0 and rn < 0:
                return "#DIV/0!"
            try:
                r = ln ** rn
            except (OverflowError, ZeroDivisionError):
                return "#DIV/0!"
            except ValueError:
                return "#VALUE!"
            if isinstance(r, complex):
                return "#VALUE!"
        else:
            return "#VALUE!"
        if not math.isfinite(r):
            return "#VALUE!"
        return r

    # -- function calls ------------------------------------------------------

    def _eval_call(self, node: _Call):
        name = node.name
        fn = _FUNCS.get(name)
        if fn is None:
            return "#NAME?"
        return fn(self, node.args)

    def _arg_value(self, node: _Node):
        """Evaluate a non-range argument node to a single value."""
        return self._eval(node)

    def _collect_numbers(self, args: list[_Node], *, count_mode: bool):
        """Shared logic for SUM/AVERAGE/MIN/MAX/COUNT.

        Returns (values, error). values is a list of floats. For COUNT,
        errors inside cells are skipped instead of returned.
        """
        values: list[float] = []
        for arg in args:
            if isinstance(arg, _Range):
                for v in self._range_values(arg):
                    if _is_err(v):
                        if count_mode:
                            continue
                        return ([], v)
                    if isinstance(v, bool):
                        continue
                    if isinstance(v, float):
                        values.append(v)
                    # text and empty cells are skipped
                continue
            v = self._eval(arg)
            if _is_err(v):
                if count_mode:
                    continue
                return ([], v)
            if isinstance(v, bool):
                values.append(1.0 if v else 0.0)
            elif isinstance(v, float):
                values.append(v)
            elif v is None:
                values.append(0.0)
            else:  # text
                if count_mode:
                    continue
                return ([], "#VALUE!")
        return (values, None)

    def _fn_sum(self, args: list[_Node]):
        values, err = self._collect_numbers(args, count_mode=False)
        if err:
            return err
        return float(sum(values))

    def _fn_average(self, args: list[_Node]):
        values, err = self._collect_numbers(args, count_mode=False)
        if err:
            return err
        if not values:
            return "#DIV/0!"
        return float(sum(values) / len(values))

    def _fn_min(self, args: list[_Node]):
        values, err = self._collect_numbers(args, count_mode=False)
        if err:
            return err
        if not values:
            return 0.0
        return float(min(values))

    def _fn_max(self, args: list[_Node]):
        values, err = self._collect_numbers(args, count_mode=False)
        if err:
            return err
        if not values:
            return 0.0
        return float(max(values))

    def _fn_count(self, args: list[_Node]):
        values, _err = self._collect_numbers(args, count_mode=True)
        return float(len(values))

    def _fn_if(self, args: list[_Node]):
        if len(args) < 2 or len(args) == 3 or len(args) > 3:
            if len(args) == 1 or len(args) > 3:
                return "#VALUE!"
        if len(args) not in (2, 3):
            return "#VALUE!"
        cond = self._eval(args[0])
        if _is_err(cond):
            return cond
        if isinstance(cond, bool):
            truthy = cond
        elif isinstance(cond, float):
            truthy = cond != 0.0
        elif cond is None:
            truthy = False
        else:
            return "#VALUE!"
        if truthy:
            return self._eval(args[1])
        if len(args) == 3:
            return self._eval(args[2])
        return False

    def _fn_concat(self, args: list[_Node]):
        parts: list[str] = []
        for arg in args:
            if isinstance(arg, _Range):
                for v in self._range_values(arg):
                    if _is_err(v):
                        return v
                    parts.append(_to_text(v))
            else:
                v = self._eval(arg)
                if _is_err(v):
                    return v
                parts.append(_to_text(v))
        return "".join(parts)

    def _fn_len(self, args: list[_Node]):
        if len(args) != 1:
            return "#VALUE!"
        v = self._eval(args[0])
        if _is_err(v):
            return v
        return float(len(_to_text(v)))

    def _fn_abs(self, args: list[_Node]):
        if len(args) != 1:
            return "#VALUE!"
        v = self._eval(args[0])
        if _is_err(v):
            return v
        n = _to_number(v)
        if _is_err(n):
            return n
        return abs(float(n))

    def _fn_round(self, args: list[_Node]):
        if len(args) not in (1, 2):
            return "#VALUE!"
        v = self._eval(args[0])
        if _is_err(v):
            return v
        n = _to_number(v)
        if _is_err(n):
            return n
        digits = 0
        if len(args) == 2:
            dv = self._eval(args[1])
            if _is_err(dv):
                return dv
            dn = _to_number(dv)
            if _is_err(dn):
                return dn
            digits = int(float(dn))  # truncated toward zero
        return _round_half_away(float(n), digits)

    "SUM": Sheet._fn_sum,
    "AVERAGE": Sheet._fn_average,
    "MIN": Sheet._fn_min,
    "MAX": Sheet._fn_max,
    "COUNT": Sheet._fn_count,
    "IF": Sheet._fn_if,
    "CONCAT": Sheet._fn_concat,
    "LEN": Sheet._fn_len,
    "ABS": Sheet._fn_abs,
    "ROUND": Sheet._fn_round,
}
